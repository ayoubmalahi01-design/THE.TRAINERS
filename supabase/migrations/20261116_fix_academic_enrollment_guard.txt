-- 1. Academic years table
CREATE TABLE IF NOT EXISTS public.academic_years (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    institution_id uuid NOT NULL
        REFERENCES public.institutions(id) ON DELETE CASCADE,
    name text NOT NULL,
    start_date date,
    end_date date,
    is_current boolean NOT NULL DEFAULT false,
    is_archived boolean NOT NULL DEFAULT false,
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT academic_year_dates_valid
        CHECK (
            start_date IS NULL
            OR end_date IS NULL
            OR start_date <= end_date
        ),

    CONSTRAINT academic_year_not_current_and_archived
        CHECK (NOT (is_current AND is_archived))
);

CREATE INDEX IF NOT EXISTS academic_years_institution_idx
ON public.academic_years(institution_id, is_current, is_archived);

CREATE UNIQUE INDEX IF NOT EXISTS academic_years_one_current_per_institution
ON public.academic_years(institution_id)
WHERE is_current IS TRUE AND is_archived IS FALSE;


-- 2. Student academic enrollments table
CREATE TABLE IF NOT EXISTS public.student_academic_enrollments (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

    student_id uuid NOT NULL
        REFERENCES public.students(id) ON DELETE CASCADE,

    academic_year_id uuid NOT NULL
        REFERENCES public.academic_years(id) ON DELETE CASCADE,

    institution_id uuid NOT NULL
        REFERENCES public.institutions(id) ON DELETE CASCADE,

    enrolled_on date,
    notes text,
    created_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT student_academic_enrollments_unique
        UNIQUE (student_id, academic_year_id)
);

CREATE INDEX IF NOT EXISTS student_academic_enrollments_year_idx
ON public.student_academic_enrollments(academic_year_id, institution_id);

CREATE INDEX IF NOT EXISTS student_academic_enrollments_student_idx
ON public.student_academic_enrollments(student_id, institution_id);


-- 3. Create the missing integrity guard function
CREATE OR REPLACE FUNCTION public.guard_student_academic_enrollment()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, pg_temp
AS $function$
BEGIN

    IF NOT EXISTS (
        SELECT 1
        FROM public.students s
        WHERE s.id = NEW.student_id
          AND s.institution_id = NEW.institution_id
    ) THEN
        RAISE EXCEPTION
            'Enrollment student must belong to the same institution';
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM public.academic_years y
        WHERE y.id = NEW.academic_year_id
          AND y.institution_id = NEW.institution_id
    ) THEN
        RAISE EXCEPTION
            'Academic year must belong to the same institution';
    END IF;

    RETURN NEW;
END;
$function$;


-- 4. Create enrollment integrity trigger if missing
DO $block$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM pg_trigger t
        WHERE t.tgrelid =
            'public.student_academic_enrollments'::regclass
          AND t.tgname =
            'student_academic_enrollment_relation_guard'
          AND NOT t.tgisinternal
    ) THEN

        CREATE TRIGGER student_academic_enrollment_relation_guard
        BEFORE INSERT OR UPDATE OF
            student_id,
            academic_year_id,
            institution_id
        ON public.student_academic_enrollments
        FOR EACH ROW
        EXECUTE FUNCTION
            public.guard_student_academic_enrollment();

    END IF;
END;
$block$;


-- 5. Updated-at trigger for academic years
DO $block$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM pg_trigger t
        WHERE t.tgrelid = 'public.academic_years'::regclass
          AND t.tgname = 'academic_years_set_updated_at'
          AND NOT t.tgisinternal
    ) THEN

        CREATE TRIGGER academic_years_set_updated_at
        BEFORE UPDATE ON public.academic_years
        FOR EACH ROW
        EXECUTE FUNCTION public.set_updated_at();

    END IF;
END;
$block$;

ON public.academic_years

TO authenticated

ON public.academic_years;
